<script>
    let phase = 0; // 0 = Standby/Tarnkappe, 1 = Wort-Abfrage läuft

    // NOTAUSSTIEG EINRICHTEN
    // Wenn Sie die obere linke Ecke für 3 Sekunden gedrückt halten, bricht der Zauber ab
    let exitTimer;
    const exitZone = document.getElementById('emergency-exit');
    const debugLog = document.getElementById('debug-log');

    exitZone.addEventListener('touchstart', () => {
        exitTimer = setTimeout(() => {
            // Tarnkappe aufheben: Hintergrund wird grau, Text wird sichtbar
            document.body.style.backgroundColor = '#222';
            debugLog.style.color = '#fff';
            debugLog.style.display = 'block';
            debugLog.innerText = "NOT-AUS: Programm abgebrochen.";
            if (recognition) recognition.stop();
        }, 3000); // 3 Sekunden halten
    });

    exitZone.addEventListener('touchend', () => {
        clearTimeout(exitTimer);
    });

    // SPRACHERKENNUNG INITIALISIEREN
    const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)();
    recognition.continuous = true;
    recognition.interimResults = false;
    recognition.lang = 'de-DE';

    recognition.onresult = (event) => {
        const text = event.results[event.results.length - 1].transcript.toLowerCase();
        
        // Test-Ausgabe im Debug-Log (falls sichtbar)
        debugLog.innerText = "Gehört: " + text;

        // WARTEN AUF DEN STARTAUSLÖSER
        if (phase === 0 && text.includes("zusammen")) {
            phase = 1;
            // Ein minimales, unsichtbares haptisches Feedback (Handy vibriert ganz kurz)
            // So wissen Sie blind, dass das Handy aktiv zuhört!
            if (navigator.vibrate) navigator.vibrate(50); 
            
            debugLog.innerText = "STARTAUSLÖSER ERKANNT! Schalte um auf Abfrage...";
            
            // HIER GEHT ES IM NÄCHSTEN SCHRITT WEITER
            starteAbfrage();
        }
    };

    // Automatischer Neustart bei Stille (Bricht Googles Standby)
    recognition.onend = () => {
        if (phase === 0 || phase === 1) {
            recognition.start();
        }
    };

    // Starten des Programms beim ersten Laden
    window.addEventListener('load', () => {
        recognition.start();
    });

    }
